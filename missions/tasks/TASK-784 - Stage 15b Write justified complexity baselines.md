```js
const health = JSON.parse(await Bun.file('/tmp/task784-health.json').text());
const section = (await Bun.file('docs/fallow-exceptions.md').text()).split('### Baselined complexity (project-health-audit)')[1].split('## Review Rules For Future Exceptions')[0];
const reasons = new Map(); let lines = [];
for (const line of section.split('\n')) {
  if (/^#### `/.test(line)) { lines = []; continue; }
  const id = line.match(/^- `([^`]+)`$/);
  if (id) { reasons.set(id[1], lines.join(' ').replace(/\s+/g, ' ').trim()); continue; }
  if (line.trim() && !line.startsWith('- ')) lines.push(line);
}
const rows = health.findings.filter(f => (/^(lib|cli|domains|scripts)\//.test(f.path) && ['high','moderate'].includes(f.severity)) || (f.path.startsWith('tests/') && f.severity === 'critical'));
const output = rows.map(f => ({metric:f.exceeded,path:f.path,line:f.line,column:f.col,name:f.name,cyclomatic:f.cyclomatic,cognitive:f.cognitive,crap:f.crap ?? null,tier:f.coverage_tier ?? 'unreported',justification:reasons.get(`${f.exceeded}/${f.path}/${f.line}/${f.col}/${f.name}`)}));
if (output.length !== 172 || reasons.size !== 172 || output.some(r => !r.justification)) throw Error('census/documentation mismatch');
```
